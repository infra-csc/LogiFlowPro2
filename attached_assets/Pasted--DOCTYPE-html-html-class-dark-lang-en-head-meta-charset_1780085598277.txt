<script>
        function toggleSubmenu(id, btn) {
            const submenu = document.getElementById(id);
            const icon = btn.querySelector('.material-symbols-outlined:last-child');
            
            // Toggle visibility
            submenu.classList.toggle('submenu-open');
            
            // Rotate icon
            if (submenu.classList.contains('submenu-open')) {
                icon.classList.add('rotate-180');
            } else {
                icon.classList.remove('rotate-180');
            }
        }

        function toggleSidebar() {
            const sidebar = document.getElementById('app-sidebar');
            const menuIcon = document.querySelector('header button span');
            
            if (sidebar.style.width === '0px') {
                sidebar.style.width = '288px'; // 72 * 4
                sidebar.style.opacity = '1';
                menuIcon.innerText = 'menu_open';
            } else {
                sidebar.style.width = '0px';
                sidebar.style.opacity = '0';
                menuIcon.innerText = 'menu';
            }
        }

        // Active state simulation for sidebar links
        const navLinks = document.querySelectorAll('nav a, nav button');
        navLinks.forEach(link => {
            link.addEventListener('click', (e) => {
                // If it's a submenu button, don't clear styling for everything
                if (link.tagName === 'BUTTON') return;

                navLinks.forEach(l => {
                    l.classList.remove('bg-secondary-container', 'text-on-secondary-container', 'font-semibold', 'scale-[0.98]');
                    l.classList.add('text-on-surface-variant');
                });
                link.classList.remove('text-on-surface-variant');
                link.classList.add('bg-secondary-container', 'text-on-secondary-container', 'font-semibold', 'scale-[0.98]');
            });
        });
    </script>
