```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

    protected $fillable = [
        'codigo', 'nome', 'descricao', 'cor', 'icone', 
        'finalidade', 'ordem_exibicao', 'ativo'

    protected $casts = [
        'ativo' => 'boolean',
        'ordem_exibicao' => 'integer'

    {
        return $this->hasMany(TipoMovimentacao::class, 'grupo_id');
    }
    
    // Scopes
    public function scopeAtivos($query)
    {
        return $query->where('ativo', true);
    }
    
    public function scopeOrdenados($query)
    {
        return $query->orderBy('ordem_exibicao')->orderBy('nome');
    }
    
    // Accessors
    public function getQuantidadeTiposAttribute()
    {
        return $this->tiposMovimentacao()->count();
    }
}
```

```php
            'finalidade' => 'required|in:OPERACIONAL,CONTROLE,TERCEIROS,AJUSTES',
```

## 📊 **2. CADASTRO DE TIPOS DE MOVIMENTAÇÃO**

### **Estrutura de Dados**
```sql
CREATE TABLE tipos_movimentacao (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  codigo VARCHAR(50) NOT NULL UNIQUE,
  nome VARCHAR(100) NOT NULL,
  grupo_id BIGINT NOT NULL,
  natureza ENUM('ENTRADA', 'SAIDA', 'TRANSFERENCIA', 'AJUSTE') NOT NULL,
  
  -- Controle de impactos (operacionais apenas)
  afeta_inventario_fisico BOOLEAN DEFAULT TRUE,
  afeta_inventario_operacional BOOLEAN DEFAULT TRUE,
  
  -- Configurações
  requer_aprovacao BOOLEAN DEFAULT FALSE,
  requer_documento BOOLEAN DEFAULT FALSE,
  permite_lote_misto BOOLEAN DEFAULT TRUE,
  
  -- Status permitidos
  status_origem JSON NULL, -- ['disponivel', 'em_evento']
  status_destino JSON NULL, -- ['disponivel', 'avariado']
  
  -- Configurações dinâmicas
  campos_obrigatorios JSON NULL, -- ['fornecedor', 'documento']
  campos_opcionais JSON NULL, -- ['observacoes', 'local']
  validacoes_especiais JSON NULL, -- regras específicas
  
  ativo BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  FOREIGN KEY (grupo_id) REFERENCES grupos_movimentacao(id)
);
```

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class TipoMovimentacao extends Model
{
    protected $table = 'tipos_movimentacao';
    
    protected $fillable = [
        'codigo', 'nome', 'grupo_id', 'natureza',
        'afeta_inventario_fisico', 'afeta_inventario_operacional',
        'requer_aprovacao', 'requer_documento', 'permite_lote_misto',
        'status_origem', 'status_destino',
        'campos_obrigatorios', 'campos_opcionais', 'validacoes_especiais',
        'ativo'
    ];
    
    protected $casts = [
        'afeta_inventario_fisico' => 'boolean',
        'afeta_inventario_operacional' => 'boolean',
        'requer_aprovacao' => 'boolean',
        'requer_documento' => 'boolean',
        'permite_lote_misto' => 'boolean',
        'ativo' => 'boolean',
        'status_origem' => 'array',
        'status_destino' => 'array',
        'campos_obrigatorios' => 'array',
        'campos_opcionais' => 'array',
        'validacoes_especiais' => 'array'
    ];
    
    // Relacionamentos
    public function grupo()
    {
        return $this->belongsTo(GrupoMovimentacao::class, 'grupo_id');
    }
    
    public function movimentacoes()
    {
        return $this->hasMany(Movimentacao::class, 'tipo_id');
    }
    
    // Scopes
    public function scopeAtivos($query)
    {
        return $query->where('ativo', true);
    }
    
    public function scopePorGrupo($query, $grupoId)
    {
        return $query->where('grupo_id', $grupoId);
    }
    
    public function scopePorNatureza($query, $natureza)
    {
        return $query->where('natureza', $natureza);
    }
    
    // Methods
    public function isCampoObrigatorio($campo)
    {
        return in_array($campo, $this->campos_obrigatorios ?? []);
    }
    
    public function getValidacaoEspecial($campo)
    {
        return $this->validacoes_especiais[$campo] ?? null;
    }
}
```

```php
<?php

namespace App\Http\Controllers;

use App\Models\GrupoMovimentacao;
use Illuminate\Http\Request;

class TipoMovimentacaoController extends Controller
{
    public function index(Request $request)
    {
        $query = TipoMovimentacao::with('grupo');
        
        // Filtros
        if ($request->has('grupo_id')) {
            $query->porGrupo($request->grupo_id);
        }
        
        if ($request->has('natureza')) {
            $query->porNatureza($request->natureza);
        }
        
        if ($request->has('ativo')) {
            $query->where('ativo', $request->boolean('ativo'));
        }
        
        $tipos = $query->orderBy('codigo')->get();
        
        return response()->json($tipos);
```

### **API para Fornecedores Recentes**
```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class FornecedorController extends Controller
{
    public function buscarPorProduto($sku)
    {
        // Fornecedores usados recentemente para este SKU
        $recentes = DB::table('movimentacao_itens')
            ->select('fornecedor_informado as nome', DB::raw('COUNT(*) as frequencia'), DB::raw('MAX(created_at) as ultima_vez'))
            ->where('sku', $sku)
            ->whereNotNull('fornecedor_informado')
            ->where('created_at', '>=', now()->subMonths(3))
            ->groupBy('fornecedor_informado')
            ->orderByDesc('frequencia')
            ->orderByDesc('ultima_vez')
            ->limit(5)
            ->get();
        
        // Fornecedores cadastrados (se existir tabela de fornecedores)
        $cadastrados = collect();
        if (Schema::hasTable('fornecedores')) {
            $cadastrados = DB::table('fornecedores')
                ->select('nome')
                ->where('ativo', true)
                ->orderBy('nome')
                ->get();
        }
        
        return response()->json([
            'recentes' => $recentes,
            'cadastrados' => $cadastrados,
            'ultimo_usado' => $recentes->first()
        ]);
    }
}
```

```vue
<script>
import ModalGrupo from './components/ModalGrupo.vue'

export default {
  name: 'GruposMovimentacao',
  components: {
    ModalGrupo
  },
  data() {
    return {
      grupos: [],
      modalAberto: false,
      grupoEdicao: null,
      loading: false
    }
  },
  async mounted() {
    await this.carregarGrupos()
  },
  methods: {
    async carregarGrupos() {
      this.loading = true
      try {
        const response = await this.$http.get('/api/grupos-movimentacao')
        this.grupos = response.data
      } catch (error) {
        this.$toast.error('Erro ao carregar grupos')
      } finally {
        this.loading = false
      }
    },
    
    abrirModal(grupo = null) {
      this.grupoEdicao = grupo
      this.modalAberto = true
    },
    
    fecharModal() {
      this.modalAberto = false
      this.grupoEdicao = null
    },
    
    async salvarGrupo(dados) {
      try {
        if (this.grupoEdicao) {
          await this.$http.put(`/api/grupos-movimentacao/${this.grupoEdicao.id}`, dados)
          this.$toast.success('Grupo atualizado com sucesso')
        } else {
          await this.$http.post('/api/grupos-movimentacao', dados)
          this.$toast.success('Grupo criado com sucesso')
        }
        
        await this.carregarGrupos()
        this.fecharModal()
      } catch (error) {
        this.$toast.error('Erro ao salvar grupo')
      }
    },
    
    editarGrupo(grupo) {
      this.abrirModal(grupo)
    },
    
    async excluirGrupo(grupo) {
      if (grupo.quantidade_tipos > 0) {
        this.$toast.warning('Não é possível excluir grupo com tipos associados')
        return
      }
      
      if (confirm('Tem certeza que deseja excluir este grupo?')) {
        try {
```

```vue
<script>
import ModalTipo from './components/ModalTipo.vue'

export default {
  name: 'TiposMovimentacao',
  components: {
    ModalTipo
  },
  data() {
    return {
      tipos: [],
      grupos: [],
      modalAberto: false,
      tipoEdicao: null,
      filtros: {
        grupo_id: '',
        natureza: '',
        ativo: '',
        busca: ''
      },
      loading: false
    }
  },
  computed: {
    tiposFiltrados() {
      let resultado = this.tipos
      
      if (this.filtros.busca) {
        const busca = this.filtros.busca.toLowerCase()
        resultado = resultado.filter(tipo => 
          tipo.codigo.toLowerCase().includes(busca) ||
          tipo.nome.toLowerCase().includes(busca)
        )
      }
      
      return resultado
    }
  },
  async mounted() {
    await Promise.all([
      this.carregarGrupos(),
      this.carregarTipos()
    ])
  },
  methods: {
    async carregarGrupos() {
      try {
        const response = await this.$http.get('/api/grupos-movimentacao')
        this.grupos = response.data
      } catch (error) {
        this.$toast.error('Erro ao carregar grupos')
      }
    },
    
    async carregarTipos() {
      this.loading = true
      try {
        const params = new URLSearchParams()
        if (this.filtros.grupo_id) params.append('grupo_id', this.filtros.grupo_id)
        if (this.filtros.natureza) params.append('natureza', this.filtros.natureza)
        if (this.filtros.ativo) params.append('ativo', this.filtros.ativo)
        
        const response = await this.$http.get(`/api/tipos-movimentacao?${params}`)
        this.tipos = response.data
      } catch (error) {
        this.$toast.error('Erro ao carregar tipos')
      } finally {
        this.loading = false
      }
    },
    
    aplicarFiltros() {
      this.carregarTipos()
    },
    
    getNaturezaClass(natureza) {
      const classes = {
        'ENTRADA': 'bg-green-100 text-green-800',
        'SAIDA': 'bg-red-100 text-red-800',
        'TRANSFERENCIA': 'bg-blue-100 text-blue-800',
        'AJUSTE': 'bg-yellow-100 text-yellow-800'
      }
      return classes[natureza] || 'bg-gray-100 text-gray-800'
    },
    
    abrirModal(tipo = null) {
      this.tipoEdicao = tipo
      this.modalAberto = true
    },
    
    fecharModal() {
      this.modalAberto = false
      this.tipoEdicao = null
    },
    
    async salvarTipo(dados) {
      try {
        if (this.tipoEdicao) {
          await this.$http.put(`/api/tipos-movimentacao/${this.tipoEdicao.id}`, dados)
          this.$toast.success('Tipo atualizado com sucesso')
        } else {
          await this.$http.post('/api/tipos-movimentacao', dados)
          this.$toast.success('Tipo criado com sucesso')
        }
        
        await this.carregarTipos()
        this.fecharModal()
      } catch (error) {
        this.$toast.error('Erro ao salvar tipo')
      }
    },
    
    editarTipo(tipo) {
      this.abrirModal(tipo)
    },
    
    duplicarTipo(tipo) {
      const duplicado = { 
        ...tipo, 
        id: null, 
        codigo: tipo.codigo + '_COPY',
        nome: tipo.nome + ' (Cópia)'
      }
      this.abrirModal(duplicado)
    },
    
    async excluirTipo(tipo) {
      if (confirm('Tem certeza que deseja excluir este tipo?')) {
        try {
          await this.$http.delete(`/api/tipos-movimentacao/${tipo.id}`)
          this.$toast.success('Tipo excluído com sucesso')
          await this.carregarTipos()
        } catch (error) {
          this.$toast.error('Erro ao excluir tipo')
        }
      }
    }
  }
}
```
